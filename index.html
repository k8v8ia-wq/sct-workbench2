      }

    // 从 Supabase 云端拉取最新数据

    async function pullFromCloud(showSuccessToast = true) {

      if (!supabaseClient) return;

      appState.cloudStatus = 'syncing';

      render();


      try {

        const { data, error } = await supabaseClient

          .from(SUPABASE_TABLE)

          .select('data, updated_at')

          .eq('id', SUPABASE_ROW_ID)

          .maybeSingle();


        if (error) throw error;


        if (data && data.data) {

          applyLoadedData(data.data);

          // 同步写入本地大容量 IndexedDB 缓存

          if (window.localforage) {

            await localforage.setItem(STORAGE_KEY, { agents: appState.agents, files: appState.files });

          }

          appState.cloudStatus = 'synced';

          appState.lastSyncTime = data.updated_at ? new Date(data.updated_at).toLocaleTimeString() : new Date().toLocaleTimeString();

          if (showSuccessToast) showToast('☁️ 已从 Supabase 云端拉取最新套餐与合同数据！');

        } else {

          // 云端表为空，自动将当前本地数据初始化推送到云端

          await pushToCloud(false);

          if (showSuccessToast) showToast('☁️ 云端初始化完成，已上传当前工作台数据！');

        }

      } catch (err) {

        console.error('Supabase pull error:', err);

        appState.cloudStatus = 'error';

        appState.cloudErrorMsg = err.message || '拉取失败，请检查表名或 RLS 策略';

        if (showSuccessToast) showToast('⚠️ 云端拉取失败：' + appState.cloudErrorMsg);

      }

      render();

    }


    // 推送当前状态到 Supabase 云端

    async function pushToCloud(showSuccessToast = false) {

      if (!supabaseClient) return;

      appState.cloudStatus = 'syncing';

      render();


      const payload = {

        agents: appState.agents,

        files: appState.files

      };


      try {

        const { error } = await supabaseClient

          .from(SUPABASE_TABLE)

          .upsert({

            id: SUPABASE_ROW_ID,

            data: payload,

            updated_at: new Date().toISOString()

          }, { onConflict: 'id' });


        if (error) throw error;


        appState.cloudStatus = 'synced';

        appState.lastSyncTime = new Date().toLocaleTimeString();

        appState.cloudErrorMsg = '';

        if (showSuccessToast) showToast('☁️ 已实时同步至 Supabase 云端！');

      } catch (err) {

        console.error('Supabase push error:', err);

        appState.cloudStatus = 'error';

        appState.cloudErrorMsg = err.message || '云端写入失败';

        showToast('⚠️ 云端保存失败（已保存在本地大容量缓存）：' + appState.cloudErrorMsg);

      }

      render();

    }

    // 解析上传的任意格式 .xlsx / .xls / .csv

    function handleXlsxFileSelected(e) {

      const file = e.target.files && e.target.files[0];

      if (!file) return;


      if (typeof XLSX === 'undefined') {

        alert('Excel 解析组件正在加载，请检查网络或稍后再试');

        return;

      }


      const reader = new FileReader();

      reader.onload = (evt) => {

        try {

          const data = new Uint8Array(evt.target.result);

          const workbook = XLSX.read(data, { type: 'array' });

          const parsedSheets = [];


          workbook.SheetNames.forEach(sheetName => {

            const ws = workbook.Sheets[sheetName];

            const matrix = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });

            const nonEmptyRows = matrix.filter(row => Array.isArray(row) && row.some(cell => String(cell).trim() !== ''));

            if (nonEmptyRows.length > 0) {

              const rawHeaders = nonEmptyRows[0].map((h, i) => String(h).trim() || `列${i + 1}`);

              const dataRows = nonEmptyRows.slice(1).map(r => rawHeaders.map((_, cIdx) => (r[cIdx] !== undefined && r[cIdx] !== null ? String(r[cIdx]).trim() : '')));

              parsedSheets.push({

                id: 'TBL-' + Date.now() + '-' + Math.random().toString(36).slice(2, 6),

                fileName: file.name,

                sheetName: sheetName,

                updatedAt: new Date().toISOString().slice(0, 10),

                headers: rawHeaders,

                rows: dataRows

              });

            }

          });


          if (parsedSheets.length === 0) {

            showToast('该 Excel 文件中没有检测到有效数据行');

            return;

          }


          appState.pendingXlsxFileName = file.name;

          appState.pendingXlsxSheets = parsedSheets;

          render();

        } catch (err) {

          console.error(err);

          alert('解析 Excel 文件失败，请确认文件未加密且为标准 .xlsx/.xls/.csv 格式');

        }

      };

      reader.readAsArrayBuffer(file);

    }
